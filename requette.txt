CREATE OR REPLACE FUNCTION generate_daily_orders()
RETURNS void AS $$
DECLARE
    r_job RECORD;
    v_order_id UUID;
    v_inserted_count INTEGER;
BEGIN
    FOR r_job IN
        SELECT
            ul.user_id,
            ul.level_id,
            l.max_order_item,  -- Nouveau nom de colonne
            l.max_product_price 
        FROM public.users_levels ul
        JOIN public.levels l ON ul.level_id = l.id
    LOOP
        v_order_id := NULL;

        -- Vérifier ou créer la commande du jour pour cet utilisateur
        SELECT id INTO v_order_id
        FROM public.orders
        WHERE user_id = r_job.user_id
          AND created_at = CURRENT_DATE;

        IF v_order_id IS NULL THEN
            INSERT INTO public.orders (user_id, is_completed, created_at)
            VALUES (r_job.user_id, false, CURRENT_DATE)
            RETURNING id INTO v_order_id;
        END IF;

        -- 1. TENTATIVE D'INSERTION DE PRODUITS (Quantité basée sur max_order_item)
        INSERT INTO public.order_items (
            order_id, product_id, is_lucky, price_at_purchase, commission, status, created_at
        )
        WITH excluded_products AS (
            SELECT oi.product_id 
            FROM public.order_items oi
            JOIN public.orders o ON oi.order_id = o.id
            WHERE o.user_id = r_job.user_id AND o.created_at = CURRENT_DATE
        ),
        selected_uniques AS (
            SELECT p.id, p.price
            FROM public.products p
            -- Note: J'ai gardé le filtre prix ici, si tu l'as supprimé de la table, retire la ligne suivante
            -- WHERE p.price <= l.max_product_price 
            WHERE p.id NOT IN (SELECT product_id FROM excluded_products)
            ORDER BY RANDOM()
            LIMIT r_job.max_order_item -- Utilisation du quota dynamique
        )
        SELECT 
            v_order_id, id, false, price, price * 0.10, 'pending', now()
        FROM selected_uniques;

        GET DIAGNOSTICS v_inserted_count = ROW_COUNT;

        -- 2. SÉCURITÉ : COMPLÉTER SI PAS ASSEZ DE PRODUITS UNIQUES TROUVÉS
        IF v_inserted_count < r_job.max_order_item THEN
            INSERT INTO public.order_items (
                order_id, product_id, is_lucky, price_at_purchase, commission, status, created_at
            )
            SELECT 
                v_order_id, p.id, false, p.price, p.price * 0.10, 'pending', now()
            FROM public.products p
            ORDER BY RANDOM()
            LIMIT (r_job.max_order_item - v_inserted_count);
        END IF;

        -- 3. ATTRIBUTION DES 'LUCKY' (20% des produits environ, ou garde 2 fixes)
        -- Ici on garde 2 lucky par niveau comme dans ton script original
        UPDATE public.order_items 
        SET is_lucky = true, 
            commission = price_at_purchase * 0.15
        WHERE id IN (
            SELECT id FROM public.order_items 
            WHERE order_id = v_order_id AND status = 'pending'
            ORDER BY created_at DESC 
            LIMIT 2
        );

    END LOOP;
END;
$$ LANGUAGE plpgsql;
